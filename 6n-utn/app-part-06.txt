async function analyzeFiles(fileList){
 const files=[...fileList].filter(f=>/\.(html?|css)$/i.test(f.name));if(!files.length){alert('No encontramos archivos HTML o CSS en la selección.');return}
 const contents=await Promise.all(files.map(async f=>({name:f.webkitRelativePath||f.name,text:await f.text(),type:/\.css$/i.test(f.name)?'css':'html'})));
 const htmlFiles=contents.filter(x=>x.type==='html'), cssFiles=contents.filter(x=>x.type==='css');const html=htmlFiles.map(x=>x.text).join('\n'),css=cssFiles.map(x=>x.text).join('\n');
 const classes=new Set();for(const m of html.matchAll(/class\s*=\s*["']([^"']+)["']/gi))m[1].split(/\s+/).filter(Boolean).forEach(c=>classes.add(c));
 const ctx={files:contents,htmlFiles,cssFiles,html,css,all:html+'\n'+css,classes};
 const results=analysisRules.map(r=>({...r,pass:!!r.test(ctx)}));renderAnalysis(results,ctx);
}
function renderAnalysis(results,ctx){
 const passed=results.filter(r=>r.pass).length,pct=Math.round(passed/results.length*100);const cbo=results.filter(r=>r.cbo),cboPass=cbo.filter(r=>r.pass).length;const sum=document.getElementById('analysisSummary');sum.className=`analysis-summary ${pct>=75&&cboPass===cbo.length?'good':'warn'}`;sum.innerHTML=`<strong>${passed} de ${results.length} verificaciones con evidencia (${pct}%)</strong><br><small>CBO detectables: ${cboPass}/${cbo.length}. ${ctx.htmlFiles.length} HTML · ${ctx.cssFiles.length} CSS. <b>Este ${pct}% es solo cumplimiento técnico detectable.</b> No pondera la calidad visual/estética, la interfaz, la coherencia del contenido, el funcionamiento observado por una persona ni la oralidad. <b>No es una nota UTN.</b></small>`;document.getElementById('techMetric').textContent=`${pct}% detectable`;
 const box=document.getElementById('analysisResults');box.innerHTML='';results.forEach(r=>{const el=document.createElement('div');el.className='check-item';el.innerHTML=`<span class="symbol">${r.pass?'✅':'⚠️'}</span><div><strong>${r.title}${r.cbo?' <span class="cbo-badge">CBO</span>':''}</strong><small>${r.pass?'Encontramos evidencia en los archivos.':r.detail}</small></div><button>${r.pass?'Repasar':'Ver ayuda'}</button>`;el.querySelector('button').addEventListener('click',()=>openLesson(r.lesson));box.appendChild(el)});
}
