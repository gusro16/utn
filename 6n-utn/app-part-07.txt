function rubricSaved(){try{return JSON.parse(localStorage.getItem('utn6n_rubric')||'{}')}catch{return {}}}
function renderRubric(){const box=document.getElementById('rubricTable');if(!box)return;const saved=rubricSaved();const filtered=rubricItems.filter(r=>rubricFilter==='all'||r.group===rubricFilter||(rubricFilter==='cbo'&&r.cbo));box.innerHTML='';filtered.forEach(r=>{const row=document.createElement('div');row.className='rubric-row';row.innerHTML=`<div class="rubric-row-head"><div><h3>${r.name}${r.cbo?'<span class="cbo-badge">CBO relacionado</span>':''}</h3><div class="rubric-meta">${r.group==='proyecto'?'PROYECTO / PRESENTACIÓN':r.group==='tecnico'?'ASPECTO TÉCNICO':'COLOQUIO'}</div></div></div><div class="rubric-levels">${r.d.map((txt,i)=>`<button class="rubric-choice level-${3-i} ${saved[r.id]===i?'selected':''}" data-rid="${r.id}" data-level="${i}"><strong>${rubricLevels[i]}</strong>${txt}</button>`).join('')}</div>`;box.appendChild(row)});box.querySelectorAll('.rubric-choice').forEach(b=>b.addEventListener('click',()=>{const x=rubricSaved();x[b.dataset.rid]=Number(b.dataset.level);localStorage.setItem('utn6n_rubric',JSON.stringify(x));renderRubric();updateRubricSummary()}));updateRubricSummary()}
function updateRubricSummary(){const s=rubricSaved();const answered=rubricItems.filter(r=>s[r.id]!==undefined);const strong=answered.filter(r=>s[r.id]<=1).length;const risk=answered.filter(r=>s[r.id]>=2).length;const cbos=rubricItems.filter(r=>r.cbo),cboAnswered=cbos.filter(r=>s[r.id]!==undefined),cboRisk=cboAnswered.filter(r=>s[r.id]>=2).length;document.getElementById('rubricAnswered').textContent=`${answered.length}/${rubricItems.length}`;document.getElementById('rubricStrong').textContent=strong;document.getElementById('rubricRisk').textContent=risk;document.getElementById('rubricCbo').textContent=cboAnswered.length?`${cboRisk} a revisar`:'Sin revisar'}
function setupRubric(){document.querySelectorAll('#rubricTabs .chip').forEach(b=>b.addEventListener('click',()=>{document.querySelectorAll('#rubricTabs .chip').forEach(x=>x.classList.remove('active'));b.classList.add('active');rubricFilter=b.dataset.rubricFilter;renderRubric()}));document.getElementById('resetRubric')?.addEventListener('click',()=>{localStorage.removeItem('utn6n_rubric');renderRubric()});renderRubric()}
