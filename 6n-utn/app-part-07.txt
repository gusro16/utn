function cleanPublicUrl6N(){
  try{
    const u=new URL(location.href);
    let changed=false;
    [...u.searchParams.entries()].forEach(([k,v])=>{
      const pair=(k+'='+v).toLowerCase();
      if(k.toLowerCase().startsWith('utm_')||pair.includes('chatgpt')||pair.includes('openai')||pair.includes('gpt')){
        u.searchParams.delete(k);changed=true;
      }
    });
    if(changed){
      const qs=u.searchParams.toString();
      history.replaceState(null,'',u.pathname+(qs?'?'+qs:'')+u.hash);
    }
  }catch(e){}
}
