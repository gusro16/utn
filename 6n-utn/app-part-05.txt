  const grid=document.getElementById('lessonGrid');grid.innerHTML='';
  const q=query.trim().toLowerCase();
  lessons.filter(l=>(filter==='all'||l.status===filter||l.area===filter)&&(!q||`${l.title} ${l.summary}`.toLowerCase().includes(q))).forEach(l=>{
