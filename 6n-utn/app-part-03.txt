function localToday(){return new Date();}
function dateOnly(s){return new Date(`${s}T12:00:00-03:00`)}

function setupNavigation(){
  document.querySelectorAll('.nav-btn,.jump').forEach(b=>b.addEventListener('click',()=>showView(b.dataset.view||b.dataset.target)));
  document.getElementById('menuBtn').addEventListener('click',()=>document.getElementById('sidebar').classList.toggle('open'));
}
function showView(id){
  document.querySelectorAll('.view').forEach(v=>v.classList.toggle('active',v.id===id));
  document.querySelectorAll('.nav-btn').forEach(b=>b.classList.toggle('active',b.dataset.view===id));
  const view=document.getElementById(id);if(!view)return;document.getElementById('viewTitle').textContent=view.dataset.title||'';
  document.getElementById('sidebar').classList.remove('open');window.scrollTo({top:0,behavior:'smooth'});
}

function renderCalendar(){
  const now=localToday();
  const diff=DELIVERY-now;const days=Math.max(0,Math.ceil(diff/86400000));
  document.getElementById('countdown').textContent=diff>0?`${days} días corridos hasta la entrega`:'Fecha de entrega alcanzada';
  const upcoming=routeSteps.find(s=>!s.noClass && dateOnly(s.date)>=new Date(now.getFullYear(),now.getMonth(),now.getDate(),0,0,0));
  document.getElementById('calendarStatus').textContent=upcoming?`Próximo hito: ${upcoming.label} · ${upcoming.title}`:'Etapa de cierre';

  const container=document.getElementById('timeline');container.innerHTML='';
  routeSteps.forEach((s,i)=>{
    const d=dateOnly(s.date);const today0=new Date(now.getFullYear(),now.getMonth(),now.getDate());
    const isDone=d<today0;const nextIndex=routeSteps.findIndex(x=>!x.noClass && dateOnly(x.date)>=today0);const current=i===nextIndex;
    const el=document.createElement('article');el.className=`timeline-item ${isDone?'done':''} ${current?'current':''} ${s.final?'final':''} ${s.noClass?'no-class':''}`;
    el.innerHTML=`<div class="timeline-date">${s.label}</div><div><h3>${s.title}</h3><p>${s.desc}</p></div><span class="timeline-tag">${s.final?'ENTREGA':s.noClass?'SIN CLASE':isDone?'REALIZADO':current?'PRÓXIMO':'SIGUE'}</span>`;
    el.addEventListener('click',()=>openRouteStep(s));container.appendChild(el);
  });
  const current=upcoming||routeSteps[routeSteps.length-1];
  document.getElementById('currentPriority').innerHTML=`<strong>${current.title}</strong><ul>${current.tasks.slice(0,3).map(t=>`<li>${t}</li>`).join('')}</ul>`;
}
